
const $TPEvent = Java.loadClass("net.minecraftforge.event.entity.EntityTeleportEvent")
const $TPCEvent = Java.loadClass("net.minecraftforge.event.entity.EntityTeleportEvent$TeleportCommand")



NativeEvents.onEvent("highest", true, $TPEvent, e => {
    /**@type {Internal.EntityTeleportEvent} */
    let event = e
    if (!event.entity.isPlayer()) { return }
    /**@type {Internal.Player} */
    let player = event.entity
    console.log(player.inventory.allItems)
})

NativeEvents.onEvent("highest", true, $TPCEvent, e => {
    /**@type {Internal.EntityTeleportEvent$TeleportCommand} */
    let event = e
    if (event.entity.level.isClientSide()) { return }
    if (!event.entity.isPlayer()) { return }
    /**@type {Internal.Player} */
    let player = event.entity
    console.log(player.inventory.allItems)
})
