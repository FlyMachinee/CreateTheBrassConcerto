
let $ContainerScreenEvent = Java.loadClass("net.minecraftforge.client.event.ContainerScreenEvent")
NativeEvents.onEvent($ContainerScreenEvent, event => {
    /**@type {Internal.ContainerScreenEvent} */
    let e = event
    e.containerScreen.getSlotUnderMouse().isSameInventory(Client.player.inventory.asContainer().)
    if (Client.isAltDown()) {
        Client.player.sendData("deleteslot",e.containerScreen.slotUnderMouse)
    }
})
